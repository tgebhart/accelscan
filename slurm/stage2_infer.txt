#!/bin/bash -l
#SBATCH --time=4:00:00
#SBATCH --signal=SIGUSR1@120
#SBATCH -p preempt-gpu
#SBATCH -A funkr
#SBATCH --nodes=1
#SBATCH --ntasks=1
#SBATCH --cpus-per-task=8
#SBATCH --mem=100gb
#SBATCH --gres=gpu:1
#SBATCH --array=0-35%16
#SBATCH --requeue
#SBATCH --mail-type=END,FAIL
#SBATCH --mail-user=gebhart@umn.edu
export PYTHONUNBUFFERED=TRUE
export OMP_NUM_THREADS=4
cd $MSIPROJECT/gebhart/projects/rfunklab/accelscan
set -a; source .env; set +a
export VLLM_CACHE_ROOT=${TMPDIR:-/tmp}/vllm_cache_${SLURM_JOB_ID}_${SLURM_ARRAY_TASK_ID:-0}
export TRITON_CACHE_DIR=$VLLM_CACHE_ROOT/triton
export TORCHINDUCTOR_CACHE_DIR=$VLLM_CACHE_ROOT/inductor
mkdir -p "$TRITON_CACHE_DIR" "$TORCHINDUCTOR_CACHE_DIR"

module load cuda/12.0
source activate accelscan

# --array is 0-based and inclusive of the last shard: 35 shards -> 0-34.
# Read the count from repack's manifest before submitting.
# Pin the registry version instead of discovering it: with a previous extraction's
# passages/ still present, discovery refuses to guess between versions (by design --
# picking the highest silently split a mentions table once). Read from the registry
# rather than hardcoded, so it follows a version bump.
REGISTRY_VERSION=$(python -c 'from accelscan.registry import load_registry; print(load_registry().version)')
echo "registry version: $REGISTRY_VERSION"

python -m accelscan.infer --shard-index $SLURM_ARRAY_TASK_ID --registry-version $REGISTRY_VERSION --model Qwen/Qwen3-14B
