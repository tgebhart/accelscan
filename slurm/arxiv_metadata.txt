# arXiv category/title/abstract metadata -> parquet on MSI S3. Supplies `field`
# (arXiv's own taxonomy), `primary_category`, and the abstracts the topic pipeline
# needs. Year is NOT taken from here -- it comes free from the arXiv id, so nothing
# in the capacity section waits on this job.
#
# Source is arXiv's own OAI-PMH endpoint, not the Kaggle snapshot: it is live rather
# than refreshed weekly, needs no credentials, and is the route arXiv documents for
# bulk *metadata* (bulk full text still comes from the requester-pays src/ tars).
# Measured at 1,300 records/page, ~3 MB/page, so ~2,080 requests and ~6.4 GB for
# ~2.7M records; at the 3s politeness delay that is roughly 2-3 hours.
#
# The harvest checkpoints its resumption token after every page, so a requeue or a
# timeout resumes instead of restarting -- rerun this identical script.
export PYTHONUNBUFFERED=TRUE
cd $MSIPROJECT/gebhart/projects/rfunklab/accelscan
set -a; source .env; set +a
source activate accelscan

OAI=/scratch.global/$USER/arxiv_oai.jsonl

python -m accelscan.scripts.harvest_arxiv_oai --out $OAI

python -m accelscan.scripts.build_arxiv_metadata --input $OAI --upload

# Check two lines of the report before trusting the field analyses:
#   "submitted before id" must be ~0% (else year_month_from_id is wrong)
#   "UNMAPPED ARCHIVES"   must not appear (else registry/arxiv_categories.yaml has a
#                         gap and those papers get a null field, silently dropping
#                         them out of every field figure)
